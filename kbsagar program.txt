  <script>
    // 1. DEFAULT BASE DATASET (Works offline immediately)
    let activeSchemes = [
      {
        id: "scheme_1",
        title: "Post Matric Scholarship for SC/ST/OBC",
        category: "Scholarship",
        incomeLimit: 250000,
        requiredDocs: ["Income Certificate", "Caste Certificate", "Marksheet"],
        successRatePercentage: 88,
        applicationUrl: "https://scholarships.gov.in/",
        lastUpdated: "2026-09-01"
      },
      {
        id: "scheme_2",
        title: "National Means Cum Merit Scholarship",
        category: "Scholarship",
        incomeLimit: 350000,
        requiredDocs: ["Income Certificate", "Class 8 Marksheet"],
        successRatePercentage: 79,
        applicationUrl: "https://scholarships.gov.in/",
        lastUpdated: "2026-09-01"
      },
      {
        id: "scheme_3",
        title: "Prime Minister Research Fellowship (PMRF)",
        category: "Higher Education",
        incomeLimit: 800000,
        requiredDocs: ["Degree Certificate", "Research Proposal"],
        successRatePercentage: 62,
        applicationUrl: "https://pmrf.in/",
        lastUpdated: "2026-08-15"
      }
    ];

    // 2. AUTOMATION: FETCH LATEST UPDATES AT STARTUP
    // Pulls updated metadata from GitHub/Public CDN endpoints into browser memory
    async function autoUpdateSchemes() {
      const statusElement = document.getElementById('sync-status');
      const REMOTE_SCHEME_API = "https://raw.githubusercontent.com/public-datasets/government-schemes/main/schemes.json";

      try {
        statusElement.innerText = "Syncing with central scheme database...";
        
        // Fetch public JSON update feed
        const response = await fetch(REMOTE_SCHEME_API, { cache: "no-cache" });
        if (response.ok) {
          const freshSchemes = await response.json();
          if (Array.isArray(freshSchemes) && freshSchemes.length > 0) {
            activeSchemes = freshSchemes;
            statusElement.innerText = `Updated! Live dataset loaded (${activeSchemes.length} active schemes available).`;
          }
        } else {
          throw new Error("Remote server unavailable");
        }
      } catch (error) {
        console.warn("Could not fetch online updates. Falling back to local offline database.", error);
        statusElement.innerText = "Running in Offline Mode (Using local built-in scheme dataset).";
      }

      // Sort schemes by success rate (Percentage of people who got benefited)
      sortAndDisplaySchemes(activeSchemes);
    }

    // 3. RANKING ENGINE (Sorted by Success Rate Percentage)
    function sortAndDisplaySchemes(schemesToDisplay) {
      const container = document.getElementById('scheme-results');
      container.innerHTML = '';

      // Sort highest success percentage first
      const sortedSchemes = [...schemesToDisplay].sort((a, b) => b.successRatePercentage - a.successRatePercentage);

      if (sortedSchemes.length === 0) {
        container.innerHTML = "<p>No matching schemes found.</p>";
        return;
      }

      sortedSchemes.forEach((scheme, index) => {
        const card = document.createElement('div');
        card.className = 'card';
        card.innerHTML = `
          <h4>#${index + 1} ${scheme.title} <span class="badge">${scheme.category}</span></h4>
          <p><strong>Success Rate (Benefited):</strong> ${scheme.successRatePercentage}%</p>
          <p><strong>Max Income Eligible:</strong> ₹${scheme.incomeLimit.toLocaleString()}</p>
          <p><strong>Required Paperwork:</strong> ${scheme.requiredDocs.join(', ')}</p>
          <a href="${scheme.applicationUrl}" target="_blank">Direct Step-by-Step Application Link</a>
        `;
        container.appendChild(card);
      });
    }

    // 4. MANUAL INPUT FILTER
    function filterSchemes() {
      const userIncome = parseFloat(document.getElementById('userIncome').value);
      if (isNaN(userIncome)) {
        alert("Please enter a valid income amount.");
        return;
      }

      const eligible = activeSchemes.filter(scheme => userIncome <= scheme.incomeLimit);
      sortAndDisplaySchemes(eligible);
    }

    // 5. OFFLINE DOCUMENT PROCESSING (Zero Storage / In-Browser Only)
    function processOfflinePdf(event) {
      const file = event.target.files[0];
      if (!file) return;

      const statusElement = document.getElementById('sync-status');
      statusElement.innerText = "Scanning PDF locally in browser memory...";

      // Standard FileReader for client-side processing without server upload
      const reader = new FileReader();
      reader.onload = function(e) {
        // Mock text scanning logic for demo purposes (e.g. PDF.js or local regex)
        console.log("PDF loaded locally. Size:", e.target.result.byteLength, "bytes");

        setTimeout(() => {
          statusElement.innerText = "PDF Scanned! Showing matching schemes (Zero data uploaded).";
          // Example: Auto-filtering based on mock scanned income threshold
          const mockScannedIncome = 200000;
          const eligible = activeSchemes.filter(scheme => mockScannedIncome <= scheme.incomeLimit);
          sortAndDisplaySchemes(eligible);
        }, 1000);
      };

      reader.readAsArrayBuffer(file);
    }

    // Automatically trigger update check when page loads
    window.addEventListener('DOMContentLoaded', autoUpdateSchemes);
  </script>
